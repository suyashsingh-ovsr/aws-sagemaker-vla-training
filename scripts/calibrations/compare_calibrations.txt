#!/usr/bin/env python3
"""
Compare calibration files between two users.
Usage: python scripts/compare_calibrations.py [user_a] [user_b]
Default: ssingh vs darren
"""

import json
import sys
from pathlib import Path

REPO_ROOT = Path(__file__).parent.parent
CALIB_ROOT = REPO_ROOT / "scripts" / "calibrations"

ARMS = [
    ("robots",        "bimanual_follower_left"),
    ("robots",        "bimanual_follower_right"),
    ("teleoperators", "bimanual_leader_left"),
    ("teleoperators", "bimanual_leader_right"),
]

JOINTS = ["shoulder_pan", "shoulder_lift", "elbow_flex", "wrist_flex", "wrist_roll", "gripper"]

# Feetech STS3215: 12-bit encoder = 4096 ticks per full 360° revolution.
# This is a hardware fact, not an assumption.
TICKS_PER_REV = 4096
TICKS_PER_DEG = TICKS_PER_REV / 360.0  # 11.377... ticks per degree

# wrist_roll is the only continuous rotation joint on SO-101.
# It can physically spin a full 360°, so a complete calibration sweep should reach
# min=0 and max=4095 (full encoder range). Any min > 0 means rotation was not
# swept all the way to the bottom stop during calibration.
# All other joints have mechanical hard stops — their true range is not known
# without a hardware datasheet, so we do NOT invent thresholds for them.
CONTINUOUS_JOINTS = {"wrist_roll"}


def load(user, arm_type, arm_name):
    path = CALIB_ROOT / user / arm_type / f"{arm_name}.json"
    with open(path) as f:
        return json.load(f)


def ticks_to_deg(ticks):
    return ticks / TICKS_PER_DEG


def pct_of_full(ticks):
    return 100.0 * ticks / TICKS_PER_REV


def compare_arm(user_a, user_b, arm_type, arm_name):
    data_a = load(user_a, arm_type, arm_name)
    data_b = load(user_b, arm_type, arm_name)

    arm_label = arm_name.replace("bimanual_", "")
    W = 96
    print(f"\n{'='*W}")
    print(f"  ARM: {arm_label.upper()}  |  {user_a} vs {user_b}")
    print(f"{'='*W}")

    # Two-row header: group label row + column label row
    g1 = f"{'':14}  {'--- ' + user_a + ' ---':^36}  {'--- ' + user_b + ' ---':^36}  {'- diff -':^16}"
    c1 = f"{'Joint':<14}  {'min':>7} {'max':>7} {'range':>7} {'% of 4096':>10}  {'min':>7} {'max':>7} {'range':>7} {'% of 4096':>10}  {'Δticks':>7} {'Δdeg':>7}"
    sep = "-" * W
    print(f"\n{g1}\n{c1}\n{sep}")

    findings = []

    for joint in JOINTS:
        j_a = data_a[joint]
        j_b = data_b[joint]

        min_a, max_a = j_a["range_min"], j_a["range_max"]
        min_b, max_b = j_b["range_min"], j_b["range_max"]
        range_a = max_a - min_a
        range_b = max_b - min_b
        delta = range_a - range_b
        delta_deg = ticks_to_deg(delta)
        pct_a = pct_of_full(range_a)
        pct_b = pct_of_full(range_b)

        print(f"{joint:<14}  "
              f"{min_a:>7} {max_a:>7} {range_a:>7} {pct_a:>9.1f}%  "
              f"{min_b:>7} {max_b:>7} {range_b:>7} {pct_b:>9.1f}%  "
              f"{delta:>+7} {delta_deg:>+6.1f}°")

        # Only flag wrist_roll — grounded in hardware (continuous joint = 0..4095)
        if joint in CONTINUOUS_JOINTS:
            issues = []
            if min_a > 0:
                issues.append(
                    f"  ** {user_a} wrist_roll: min={min_a} ticks, not 0 "
                    f"=> missed {ticks_to_deg(min_a):.1f}° at bottom of rotation"
                )
            if min_b > 0:
                issues.append(
                    f"  ** {user_b} wrist_roll: min={min_b} ticks, not 0 "
                    f"=> missed {ticks_to_deg(min_b):.1f}° at bottom of rotation"
                )
            if issues:
                for issue in issues:
                    print(issue)
                findings.append((joint, min_a, min_b, range_a, range_b))

    return findings


def main():
    user_a = sys.argv[1] if len(sys.argv) > 1 else "ssingh"
    user_b = sys.argv[2] if len(sys.argv) > 2 else "darren"

    print(f"\nCALIBRATION COMPARISON: {user_a.upper()} vs {user_b.upper()}")
    print(f"Calibration root: {CALIB_ROOT}")

    all_findings = []
    for arm_type, arm_name in ARMS:
        findings = compare_arm(user_a, user_b, arm_type, arm_name)
        for f in findings:
            all_findings.append((arm_name, *f))

    print(f"\n{'='*80}")

    print(f"{'='*80}")
    if all_findings:
        for arm_name, joint, min_a, min_b, range_a, range_b in all_findings:
            arm_label = arm_name.replace("bimanual_", "")
            if min_a > 0:
                print(f"  {arm_label:<22} {user_a}: wrist_roll min={min_a} → "
                      f"captured {ticks_to_deg(range_a):.1f}° of 360°  "
                      f"(missing {ticks_to_deg(min_a):.1f}° at bottom)")
            if min_b > 0:
                print(f"  {arm_label:<22} {user_b}: wrist_roll min={min_b} → "
                      f"captured {ticks_to_deg(range_b):.1f}° of 360°  "
                      f"(missing {ticks_to_deg(min_b):.1f}° at bottom)")
    else:


        print(f"""
WHAT THESE NUMBERS MEAN
-----------------------
  range_min    Encoder tick recorded at the joint's minimum position during calibration.
  range_max    Encoder tick recorded at the joint's maximum position during calibration.
  range        Span in ticks: range_max - range_min.
               Larger range = wider motion the robot knows about.
  %enc         Percentage of the full encoder range (4096 ticks) that was captured.
               100% means the joint was swept across the full encoder resolution.
  Δticks       (user_a range) minus (user_b range).
               Negative = user_a captured fewer ticks = less motion range.
  Δdeg         Same delta converted to degrees (1 tick = {1/TICKS_PER_DEG:.4f}°).

  
""")


if __name__ == "__main__":
    main()
